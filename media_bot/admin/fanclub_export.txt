(() => {
  if (!["https://takanekofc.com", "https://www.takanekofc.com"].includes(location.origin)) {
    throw new Error("Open your signed-in Takaneko Fanclub page first.");
  }
  if (document.querySelector("#takaneko-login-export")) return;

  function bindExporter() {
    const field = document.querySelector("input");
    const kind = document.querySelector("select");
    const status = document.querySelector('[role="status"]');
    const clearSelection = () => document.querySelector("textarea")?.remove();
    function showHint() {
      clearSelection();
      field.value = "";
      status.textContent = "";
      document.querySelector("#hint").textContent = kind.value === "refresh"
        ? "In DevTools, open Application → Local Storage → the Fanclub origin. Copy your refreshToken value."
        : "In DevTools Network, select api.takanekofc.com/auth/refresh. Copy the login cookie as name=value. Separate multiple pairs with a semicolon.";
    }
    function makeSession() {
      const value = field.value.trim();
      if (!value) throw new Error("Paste the selected value first.");
      if (kind.value === "refresh") {
        if (!/^[!-~]{1,16384}$/.test(value) || /^Cookie:/i.test(value))
          throw new Error("Paste only the refreshToken value.");
        return {refreshToken: value};
      }
      const pairs = value.replace(/^Cookie:\s*/i, "").split(";");
      if (pairs.length > 1000) throw new Error("Select only the Fanclub API login cookies.");
      const cookies = pairs.map(pair => {
        const match = /^([!#$%&'*+.^_`|~0-9A-Za-z-]{1,256})=([\x21-\x3A\x3C-\x7E]{1,16384})$/.exec(pair.trim());
        if (!match) throw new Error("Use name=value pairs from the refresh request.");
        return {name: match[1], value: match[2], domain: "api.takanekofc.com", path: "/auth/refresh"};
      });
      return {cookies};
    }
    kind.addEventListener("change", showHint);
    field.addEventListener("input", clearSelection);
    document.querySelector("button").addEventListener("click", () => {
      clearSelection();
      let json;
      try {
        const session = makeSession();
        json = JSON.stringify(session);
        // The broker counts spaces after JSON separators when validating the size.
        const spaces = session.cookies ? 8 * session.cookies.length : 1;
        if (json.length + spaces > 44 * 1024) throw new Error("Select only the login values for the Fanclub API.");
      } catch (error) {
        status.textContent = error.message;
        return;
      }
      // Copy the selection inside this isolated frame using the browser's clipboard command.
      const selection = document.createElement("textarea");
      selection.setAttribute("aria-label", "Login JSON selected for copying");
      selection.readOnly = true;
      selection.value = json;
      document.body.append(selection);
      selection.select();
      let copied = false;
      try { copied = document.execCommand("copy"); } catch {}
      if (copied) {
        selection.remove();
        field.value = "";
        status.textContent = "Copied. Paste into Fanclub login (JSON) in your bot console, then save.";
      } else {
        field.value = "";
        selection.focus();
        selection.select();
        status.textContent = "Press Ctrl+C or ⌘C to copy the selected JSON, then close this dialog.";
      }
    });
    showHint();
    document.querySelector("#loading").remove();
    document.querySelector("main").hidden = false;
  }

  const dialog = document.createElement("dialog");
  dialog.id = "takaneko-login-export";
  dialog.setAttribute("aria-label", "Export Fanclub login");
  const frame = document.createElement("iframe");
  frame.title = "Format selected Fanclub login data";
  // Omit allow-same-origin: the host page cannot read the input or clipboard selection.
  frame.setAttribute("sandbox", "allow-scripts");
  frame.setAttribute("allow", "clipboard-write *");
  frame.style.cssText = "width:min(32rem,72vw);height:20rem;border:0;display:block";
  const contents = `<!doctype html><html lang="en"><meta charset="utf-8">
    <meta http-equiv="Content-Security-Policy" content="default-src 'none'; script-src 'unsafe-inline'; form-action 'none'; base-uri 'none'">
    <title>Export Fanclub login</title><h2>Export Fanclub login</h2>
    <p id="loading">If the helper does not open, use the manual login JSON formats in your bot console.</p><main hidden>
    <label>Login data <select><option value="refresh">Refresh token</option>
    <option value="cookies">API cookies</option></select></label><p id="hint"></p>
    <label>Paste selected value <input type="password" autocomplete="off" spellcheck="false" aria-describedby="hint"></label>
    <p><button type="button">Copy login JSON</button></p><p role="status"></p></main>
    <script>(${bindExporter})();<\/script></html>`;
  try { frame.srcdoc = contents; }
  catch { throw new Error("The site blocked the helper. Use manual login JSON in your bot console."); }
  const close = document.createElement("button");
  close.type = "button";
  close.textContent = "Close";
  close.addEventListener("click", () => dialog.close());
  dialog.addEventListener("close", () => dialog.remove());
  dialog.append(frame, close);
  document.body.append(dialog);
  dialog.showModal();
})();
